cmake_minimum_required(VERSION 3.20)
project(SubVulpesRuntimeSpikes LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use MSVC x86 for runtime spikes.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/arch:IA32>" "$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>")
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
get_filename_component(SV_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(RUNTIMES "${SV_ROOT}/build/runtimes")
set(LUA "${RUNTIMES}/lua-5.4.9/src")
file(GLOB LUA_SOURCES "${LUA}/*.c")
list(REMOVE_ITEM LUA_SOURCES "${LUA}/lua.c" "${LUA}/luac.c")
add_library(spike-lua STATIC ${LUA_SOURCES})
target_include_directories(spike-lua PUBLIC "${LUA}")
add_executable(lua-spike lua_spike.c)
target_link_libraries(lua-spike PRIVATE spike-lua)
set(BUILD_WASI "none")
add_subdirectory("${RUNTIMES}/wasm3-0.5.0/source" "${CMAKE_BINARY_DIR}/wasm3")
target_compile_definitions(m3 PRIVATE d_m3MaxLinearMemoryPages=2)
add_executable(wasm3-spike wasm3_spike.c)
target_link_libraries(wasm3-spike PRIVATE m3)
set(WAMR_BUILD_TARGET "X86_32")
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_BUILD_THREAD_MGR 1)
set(WAMR_BUILD_REF_TYPES 0)
add_subdirectory("${RUNTIMES}/wasm-micro-runtime-WAMR-2.4.5/product-mini/platforms/windows"
    "${CMAKE_BINARY_DIR}/wamr")
add_executable(wamr-spike wamr_spike.c)
target_compile_definitions(wamr-spike PRIVATE WASM_RUNTIME_API_EXTERN=)
target_compile_options(vmlib PRIVATE "$<$<COMPILE_LANGUAGE:ASM_MASM>:/safeseh>")
target_include_directories(wamr-spike PRIVATE
    "${RUNTIMES}/wasm-micro-runtime-WAMR-2.4.5/core/iwasm/include")
target_link_libraries(wamr-spike PRIVATE vmlib)

set_property(SOURCE lua_spike.c wasm3_spike.c wamr_spike.c APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/spike.h")
